// Employee is an employee record (Employee in problem.yaml).
type Employee struct {
	Id           int
	Importance   int
	Subordinates []int
}

// hcDecEmployee0 reads [id, importance, [subordinate ids]].
func hcDecEmployee0(raw json.RawMessage) *Employee {
	parts := hcRaw1(raw)
	if len(parts) != 3 {
		panic(hcInputError{fmt.Sprintf("an employee must be [id, importance, [subordinate ids]], got %s", raw)})
	}
	return &Employee{Id: hcDec0(parts[0], hcParseInt), Importance: hcDec0(parts[1], hcParseInt), Subordinates: hcDec1(parts[2], hcParseInt)}
}

func hcDecEmployee1(raw json.RawMessage) []*Employee {
	parts := hcRaw1(raw)
	out := make([]*Employee, len(parts))
	for i, p := range parts {
		out[i] = hcDecEmployee0(p)
	}
	return out
}

func hcEncEmployee0(e *Employee) any {
	if e == nil {
		return nil
	}
	return []any{e.Id, e.Importance, e.Subordinates}
}

func hcEncEmployee1(es []*Employee) any {
	out := make([]any, len(es))
	for i, e := range es {
		out[i] = hcEncEmployee0(e)
	}
	return out
}
